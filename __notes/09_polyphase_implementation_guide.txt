      // -----------------------------------------------------------
      // Detector
      // -----------------------------------------------------------
      bool is_polyphase_friendly(
          uint32_t in_channels,
          uint32_t out_channels,
          std::array<uint32_t, 2> kernel_size,
          std::array<uint32_t, 2> stride,
          uint32_t input_width,
          uint32_t output_width) {
          const uint32_t S = stride[1];
          const uint32_t K = kernel_size[1];
          if (S < 2)              return false;     // not a transpose-strided op
          if (K < 4 * S)          return false;     // savings too small
          if (in_channels > 32)   return false;     // already efficient
          if (output_width < 16 * S) return false;  // amortization too small

          // Optionally exclude H>1 cases for the first version.
          if (kernel_size[0] != 1 || stride[0] != 1) return false;

          return true;
      }

      // -----------------------------------------------------------
      // Polyphase weight shuffle
      // -----------------------------------------------------------
      // Given weights w[in, out, K_h=1, K_w], stride S, we build S
      // kernels each of shape [in, out, 1, K_p] where K_p = ceil(K_w / S).
      // For phase p in 0..S-1:
      //     w_p[:, :, 0, k'] = w[:, :, 0, k' * S + p]
      //                        (zero-pad if k' * S + p >= K_w)
      ttnn::Tensor shuffle_weights_polyphase(
          const ttnn::Tensor& weight_tensor,
          uint32_t stride_w,
          bool mirror_kernel) {
          // 1. Read the weight into a row-major tensor on host.
          auto weight_host = weight_tensor.cpu().to_layout(Layout::ROW_MAJOR);
          auto shape = weight_host.logical_shape();        // [in, out, 1, K]
          const uint32_t in   = shape[0];
          const uint32_t out  = shape[1];
          const uint32_t K    = shape[3];
          const uint32_t S    = stride_w;
          const uint32_t K_p  = (K + S - 1) / S;

          // 2. Optional kernel mirror (parity with PyTorch).
          //    Reuse the existing helper if mirror_kernel == true.
          ttnn::Tensor pre_mirrored = mirror_kernel
              ? transform_weights_for_conv_transpose2d(weight_host, /*mirror=*/true)
              : weight_host;

          // 3. Build the polyphase tensor of shape [in, out, 1, S * K_p].
          //    Layout: phase p occupies columns [p*K_p, (p+1)*K_p).
          ttnn::Tensor out_tensor = empty_like(pre_mirrored, /*K_w=*/S * K_p);

          auto* src = pre_mirrored.data_ptr<float>();   // adjust dtype as needed
          auto* dst = out_tensor.data_ptr<float>();

          for (uint32_t i = 0; i < in; ++i) {
              for (uint32_t o = 0; o < out; ++o) {
                  for (uint32_t p = 0; p < S; ++p) {
                      for (uint32_t kp = 0; kp < K_p; ++kp) {
                          uint32_t k_orig = kp * S + p;
                          uint32_t src_idx = ((i * out + o) * 1 + 0) * K + k_orig;
                          uint32_t dst_idx = ((i * out + o) * 1 + 0) * (S * K_p)
                                              + p * K_p + kp;
                          dst[dst_idx] = (k_orig < K) ? src[src_idx] : 0.0f;
                      }
                  }
              }
          }

          return out_tensor;     // caller reshapes to per-phase as needed
      }

  WHY:
      `is_polyphase_friendly` lets the dispatcher decide cheaply.
      `shuffle_weights_polyphase` builds the S sub-kernels in one tensor.

  TYPE NOTE:
      The pseudocode uses `float` for clarity; the real implementation
      should template on dtype or branch on weight_host.dtype().
      Use ttnn::Tensor's existing host-side accessors instead of raw
      data_ptr if you want to keep dtype-agnostic.

      // SPDX-FileCopyrightText: (c) 2026 Tenstorrent USA, Inc.
      // SPDX-License-Identifier: Apache-2.0

      #include "conv_transpose2d_polyphase.hpp"
      #include "prepare_conv_transpose2d_weights.hpp"

      #include <ttnn/operations/conv/conv2d/conv2d.hpp>
      #include <ttnn/operations/data_movement/pad/pad.hpp>
      #include <ttnn/operations/data_movement/reshape/reshape.hpp>
      #include <ttnn/operations/data_movement/transpose/transpose.hpp>
      #include <ttnn/operations/data_movement/slice/slice.hpp>

      namespace ttnn::operations::conv::conv_transpose2d {

      ConvTranspose2dResult conv_transpose2d_POLYPHASE(
          const ttnn::Tensor& input_tensor,
          const ttnn::Tensor& weight_tensor,
          MeshDevice* device,
          uint32_t in_channels,
          uint32_t out_channels,
          uint32_t batch_size,
          uint32_t input_height,    // expected = 1 in v1
          uint32_t input_width,     // = T
          std::array<uint32_t, 2> kernel_size,
          std::array<uint32_t, 2> stride,
          std::variant<std::array<uint32_t, 2>, std::array<uint32_t, 4>> padding,
          std::array<uint32_t, 2> output_padding,
          std::array<uint32_t, 2> dilation,
          uint32_t groups,
          const std::optional<const DataType>& dtype,
          const std::optional<const ttnn::Tensor>& bias_tensor,
          const std::optional<const Conv2dConfig>& conv_config_,
          const std::optional<const DeviceComputeKernelConfig>& compute_config_,
          const std::optional<const MemoryConfig>& memory_config_,
          bool mirror_kernel) {

          // -------- 1. Geometry --------
          const uint32_t T   = input_width;
          const uint32_t K   = kernel_size[1];
          const uint32_t S   = stride[1];
          const uint32_t K_p = (K + S - 1) / S;
          const uint32_t T_out =
              (T - 1) * S - /*pad*/0 + (K - 1) + /*out_pad*/0 + 1;
          const uint32_t T_per_phase = (T_out + S - 1) / S;
          const uint32_t T_padded = T_per_phase + K_p - 1;
          const uint32_t right_pad = T_padded - T;

          // -------- 2. Pre-shuffled weights (cache by weight ptr in P2)
          ttnn::Tensor weights_polyphase = shuffle_weights_polyphase(
              weight_tensor, S, mirror_kernel);
          // shape: [in, out, 1, S * K_p]

          // -------- 3. Pad input on the W axis (right pad only for v1)
          ttnn::Tensor x_padded = ttnn::pad(
              input_tensor,
              std::vector<std::pair<uint32_t, uint32_t>>{
                  {0, 0},                  // batch
                  {0, 0},                  // H
                  {0, right_pad},          // W
                  {0, 0},                  // C
              },
              /*pad_value=*/0.0f);

          // -------- 4. Run S sub-convs (Option C: launch in a loop)
          //              For P2: switch to grouped/batched single launch.
          std::vector<ttnn::Tensor> phase_outputs;
          phase_outputs.reserve(S);

          for (uint32_t p = 0; p < S; ++p) {
              // Slice the per-phase kernel out of weights_polyphase.
              // shape: [in, out, 1, K_p] for this phase.
              ttnn::Tensor w_p = ttnn::slice(
                  weights_polyphase,
                  /*starts=*/{0, 0, 0, p * K_p},
                  /*ends=*/{in_channels, out_channels, 1, (p + 1) * K_p});

              // Conv2d (NORMAL conv, not transpose).
              auto y_p = ttnn::conv2d(
                  x_padded,
                  w_p,
                  device,
                  in_channels,
                  out_channels,
                  batch_size,
                  /*input_h=*/1,
                  /*input_w=*/T_padded,
                  /*kernel=*/{1, K_p},
                  /*stride=*/{1, 1},
                  /*padding=*/std::array<uint32_t, 4>{0, 0, 0, 0},
                  /*dilation=*/{1, 1},
                  /*groups=*/1,
                  dtype,
                  /*bias=*/std::nullopt,        // bias added at end
                  conv_config_,
                  compute_config_,
                  memory_config_,
                  /*return_output_dim=*/false,
                  /*return_weights_and_bias=*/false);
              phase_outputs.push_back(std::get<0>(y_p));
          }

          // -------- 5. Interleave the S phase outputs --------
          // Each y_p is [B, 1, T_per_phase, C_out].
          // Stack along a new axis and flatten:
          //     stack -> [B, 1, T_per_phase, S, C_out]
          //     reshape -> [B, 1, T_per_phase * S, C_out]
          //     slice -> [B, 1, T_out, C_out]
          ttnn::Tensor stacked = ttnn::stack(phase_outputs, /*dim=*/3);
          ttnn::Tensor reshaped = ttnn::reshape(
              stacked,
              {batch_size, 1, T_per_phase * S, out_channels});
          ttnn::Tensor result = (T_per_phase * S == T_out)
              ? reshaped
              : ttnn::slice(reshaped,
                            /*starts=*/{0, 0, 0, 0},
                            /*ends=*/{batch_size, 1, T_out, out_channels});

          // -------- 6. Add bias (if any) --------
          if (bias_tensor.has_value()) {
              result = ttnn::add(result, bias_tensor.value());
          }

          return ConvTranspose2dResult{
              result,
              {1, T_out},                             // output_dim
              weights_polyphase,                      // prepared weights
              bias_tensor,                            // prepared bias
          };
      }

      }  // namespace ttnn::operations::conv::conv_transpose2d

  WHY:
      This is the core of the new path. It:
        1. Computes per-phase shapes.
        2. Shuffles weights (one-time per layer).
        3. Pads x on the right with the small number of zeros needed.
        4. Runs S small conv2ds (using the EXISTING ttnn::conv2d).
        5. Interleaves the outputs into the final tensor.
        6. Adds bias.

      Notice: NO halo, NO is_transpose, NO zero interleaving. The
      conv2d call is a totally normal stride-1 conv.

      """Tests for the polyphase fast path of ttnn.conv_transpose2d."""

      import pytest
      import torch
      import torch.nn.functional as F
      import ttnn


      def _pcc(a: torch.Tensor, b: torch.Tensor) -> float:
          a, b = a.flatten().float(), b.flatten().float()
          return torch.corrcoef(torch.stack([a, b]))[0, 1].item()


      @pytest.mark.parametrize(
          "in_ch,out_ch,k,s,T,with_bias",
          [
              # Friendly shapes that SHOULD trigger polyphase:
              (1, 1, 16, 8, 32, False),    # envelope_like
              (1, 1, 32, 8, 32, False),
              (1, 1, 640, 160, 96, False), # iSTFT envelope (the prize)
              (8, 1, 640, 160, 96, False),
              (32, 1, 640, 160, 96, False),
              # Edge cases:
              (1, 1, 17, 8, 32, False),    # K not divisible by S
              (1, 1, 16, 8, 32, True),     # with bias
          ],
      )
      def test_polyphase_path(device, in_ch, out_ch, k, s, T, with_bias):
          torch.manual_seed(0)

          x = torch.randn(1, in_ch, T, dtype=torch.float32)
          w = torch.randn(in_ch, out_ch, k, dtype=torch.float32)
          b = torch.randn(out_ch) if with_bias else None
          ref = F.conv_transpose1d(x, w, bias=b, stride=s).contiguous()

          x_nhwc = x.transpose(1, 2).unsqueeze(1).contiguous()
          x_tt = ttnn.from_torch(
              x_nhwc, dtype=ttnn.bfloat16, layout=ttnn.ROW_MAJOR_LAYOUT,
              device=device, memory_config=ttnn.DRAM_MEMORY_CONFIG)
          w_tt = ttnn.from_torch(w.unsqueeze(2), dtype=ttnn.bfloat16)
          b_tt = ttnn.from_torch(b.reshape(1, 1, 1, -1), dtype=ttnn.bfloat16) \
                  if with_bias else None

          y_tt, _, _ = ttnn.conv_transpose2d(
              input_tensor=x_tt,
              weight_tensor=w_tt,
              in_channels=in_ch,
              out_channels=out_ch,
              device=device,
              bias_tensor=b_tt,
              kernel_size=(1, k),
              stride=(1, s),
              padding=(0, 0),
              output_padding=(0, 0),
              batch_size=1,
              input_height=1,
              input_width=T,
              groups=1,
              mirror_kernel=True,
              return_output_dim=True,
              return_weights_and_bias=True,
              dtype=ttnn.bfloat16,
          )

          y = ttnn.to_torch(y_tt).squeeze().transpose(0, 1).contiguous()
          assert y.shape[-1] == ref.shape[-1], f"shape: {y.shape} vs {ref.shape}"
          pcc = _pcc(ref, y)
          assert pcc > 0.99, f"PCC {pcc:.4f} < 0.99"

  WHY:
      Confirms the new path produces correct results and unblocks the
      iSTFT shapes.

================================================================================
  CHAPTER 9 - GOTCHAS YOU WILL HIT
================================================================================

  - `ttnn::stack` may not exist with that exact name. Use
    `ttnn::experimental::stack` or build it from `concat` + `reshape`.

  - `ttnn::pad` requires the input to be in a specific layout. For NHWC
    ROW_MAJOR you may need a `to_layout` call first.

  - `ttnn::conv2d`'s public API changes occasionally; double-check the
    argument order against `conv2d.hpp` before you build.

  - Bias addition: applying bias inside each sub-conv would multiply
    bias by S (because each phase contributes). Add bias ONCE at the end.

  - Output size off-by-one: if `T_out` is not divisible by S, the
    interleave produces `T_per_phase * S >= T_out` outputs. Slice
    down to T_out at the end.

  - `mirror_kernel`: must be applied BEFORE the polyphase shuffle, not
    after.

  - Dtype mismatch in interleave: the per-phase outputs may have
    different memory layouts. Bring them all into a common
    (sharded or interleaved) memory_config before stacking.

  - Tile alignment: each per-phase conv2d output is padded up to
    TILE_HEIGHT. The slice at the end must respect this.

  - Tests: when stride doesn't divide K cleanly, the last phase has
    a tap with k_orig >= K -> set those weights to 0. The fp32 ref in
    PyTorch handles this automatically because the equivalent samples
    are out-of-bounds. Verify this corner.
