================================================================================
  fft/  —  HOW TO RUN  (fp32 multi-core FFT, 1..64 cores, N up to 65536)
  Run every command from the tt-metal repo root.
================================================================================


--------------------------------------------------------------------------------
0.  ONE-TIME ENV SETUP
--------------------------------------------------------------------------------

source python_env/bin/activate
export ARCH_NAME=wormhole_b0
export TT_METAL_HOME=$(pwd)


--------------------------------------------------------------------------------
1.  BUILD  (first time only, plus after editing kernels/host)
--------------------------------------------------------------------------------

# Configure once
cmake -B build -DBUILD_PROGRAMMING_EXAMPLES=ON

# Build all FFT executables in one shot
cmake --build build \
    --target metal_example_fft_test \
             metal_example_fft_demo \
             metal_example_fft_vs_torch \
             metal_example_fft_benchmark \
             my_fft_app \
    -j


--------------------------------------------------------------------------------
2.  RUN THE KERNEL  (correctness + timing for N = 16 .. 65536)
--------------------------------------------------------------------------------

# 15 tests — impulse / DC / random — auto-scales 1..64 cores.
ARCH_NAME=wormhole_b0 TT_METAL_CLEAR_JIT_CACHE=1 \
    ./build/programming_examples/fft/metal_example_fft_test

# Without clearing the JIT cache (faster on repeat runs)
ARCH_NAME=wormhole_b0 \
    ./build/programming_examples/fft/metal_example_fft_test

# Expected: every line should say [PASS] with rel error < 2e-3.


--------------------------------------------------------------------------------
3.  RUN THE BENCHMARK  (plan-cache speedup, fft() called many times)
--------------------------------------------------------------------------------

# Default: N=4096, 100 iterations
ARCH_NAME=wormhole_b0 \
    ./build/programming_examples/fft/metal_example_fft_benchmark

# Pick your own N (power of 2, [2, 65536]) and iteration count
ARCH_NAME=wormhole_b0 \
    ./build/programming_examples/fft/metal_example_fft_benchmark 1024 100

ARCH_NAME=wormhole_b0 \
    ./build/programming_examples/fft/metal_example_fft_benchmark 16384 100

ARCH_NAME=wormhole_b0 \
    ./build/programming_examples/fft/metal_example_fft_benchmark 65536 100

# Output shows: first call (build cost) vs cached avg (steady state) + speedup.


--------------------------------------------------------------------------------
4.  RUN THE DEMO  (PyTorch-style worked examples)
--------------------------------------------------------------------------------

ARCH_NAME=wormhole_b0 TT_METAL_CLEAR_JIT_CACHE=1 \
    ./build/programming_examples/fft/metal_example_fft_demo

# Prints FFT of:
#   [10, 20, 30, 40]
#   [1, j, -1, -j]
#   cos(2*pi*n/8)
#   ramp [0..1023]


--------------------------------------------------------------------------------
5.  COMPARE  vs  torch.fft.fft   (Python script, accuracy report)
--------------------------------------------------------------------------------

# Default (N=1024, seed=42, threshold 2e-3)
python tt_metal/programming_examples/fft/compare_with_torch.py

# Pick N
python tt_metal/programming_examples/fft/compare_with_torch.py --N 4096
python tt_metal/programming_examples/fft/compare_with_torch.py --N 8192
python tt_metal/programming_examples/fft/compare_with_torch.py --N 16384
python tt_metal/programming_examples/fft/compare_with_torch.py --N 65536

# Different RNG seed
python tt_metal/programming_examples/fft/compare_with_torch.py --N 8192 --seed 7

# Tighter threshold
python tt_metal/programming_examples/fft/compare_with_torch.py --N 1024 --tol 1e-4

# Keep the temp input/output text files
python tt_metal/programming_examples/fft/compare_with_torch.py --N 256 --keep

# Custom binary path
python tt_metal/programming_examples/fft/compare_with_torch.py \
    --N 4096 \
    --bin ./build/programming_examples/fft/metal_example_fft_vs_torch

# Output shows: max abs err, max rel err, mean err, SNR (dB), worst 5 bins,
# and PASS/FAIL.


--------------------------------------------------------------------------------
6.  DFT (O(N^2)) vs torch.fft.fft   (built into the test)
--------------------------------------------------------------------------------

# The test harness already does this:
#   - For N <= 4096: it uses an O(N^2) reference DFT (gold standard).
#   - For N >  4096: it uses an O(N log N) double-precision FFT
#                   (still way more accurate than the device).
#
# Just run:
ARCH_NAME=wormhole_b0 TT_METAL_CLEAR_JIT_CACHE=1 \
    ./build/programming_examples/fft/metal_example_fft_test
#
# Each [PASS] line shows: abs err, rel err, ms, P (cores).
# rel err comes from comparing tt-metal output against the DFT/FFT reference.


--------------------------------------------------------------------------------
7.  RAW DUMPER  (writes input + tt-metal output to text files)
--------------------------------------------------------------------------------

# Used internally by compare_with_torch.py, but you can run it directly:
ARCH_NAME=wormhole_b0 \
    ./build/programming_examples/fft/metal_example_fft_vs_torch \
    <N> [seed] [in_path] [out_path]

# Example: N=1024
ARCH_NAME=wormhole_b0 \
    ./build/programming_examples/fft/metal_example_fft_vs_torch \
    1024 42 /tmp/in.txt /tmp/out.txt

# Each file: N lines of "real imag" in scientific notation.


--------------------------------------------------------------------------------
8.  YOUR OWN SIGNAL  (edit, build, run)
--------------------------------------------------------------------------------

# Edit the signal inside main():
#   tt_metal/programming_examples/fft/my_fft_app.cpp
cmake --build build --target my_fft_app -j
ARCH_NAME=wormhole_b0 TT_METAL_CLEAR_JIT_CACHE=1 \
    ./build/programming_examples/fft/my_fft_app


--------------------------------------------------------------------------------
9.  USE THE FFT FROM ANY C++ FILE  (PyTorch-style API)
--------------------------------------------------------------------------------

# In your .cpp:
#
#   #include "tt_metal/programming_examples/fft/fft_host.cpp"
#   using namespace fft_example;
#
#   auto md = MeshDevice::create_unit_mesh(0);
#   std::vector<float> signal = {10.f, 20.f, 30.f, 40.f};
#   auto spectrum = fft(md, signal);
#
#   // or complex input:
#   std::vector<std::complex<float>> cx = { {1,0}, {0,1}, {-1,0}, {0,-1} };
#   auto spec2 = fft(md, cx);
#
# Requirement: signal.size() is a power of 2 in [2, 65536].
# Equivalent to torch.fft.fft(signal).
# First call for a new N pays the program-build cost (~10-25 ms);
# every subsequent call hits the plan cache (~1-3 ms device-time).
